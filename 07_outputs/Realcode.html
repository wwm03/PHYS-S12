// ============================================================
// SENSOR FUNCTION
// ============================================================

long tx_rx() {
  int read_high;
  int read_low;
  long sum = 0;

  const int N_samples = 100;

  for (int i = 0; i < N_samples; i++) {
    digitalWrite(tx_pin, HIGH);
    read_high = analogRead(analog_pin);

    delayMicroseconds(100);

    digitalWrite(tx_pin, LOW);
    read_low = analogRead(analog_pin);

    sum += read_high - read_low;
  }

  return sum;
}

// ============================================================
// CALCULATION
// ============================================================

float rawToMl(long raw) {
  if (!calibrated || fullRaw == emptyRaw) {
    return 0.0f;
  }

  float ml =
    BOTTLE_SIZE_ML *
    (float)(raw - emptyRaw) /
    (float)(fullRaw - emptyRaw);

  if (ml < 0.0f) ml = 0.0f;
  if (ml > BOTTLE_SIZE_ML) ml = BOTTLE_SIZE_ML;

  return ml;
}

void processWaterTracking() {
  if (!calibrated) return;

  if (!trackingStarted) {
    referenceWater = water;
    lastDrinkTime = millis();
    trackingStarted = true;
    return;
  }

  if (water > referenceWater + REFILL_THRESHOLD_ML) {
    referenceWater = water;
    Serial.println("Refill detected");
    return;
  }

  if (referenceWater - water >= DRINK_THRESHOLD_ML) {
    float drank = referenceWater - water;

    if (drank <= BOTTLE_SIZE_ML) {
      totalDrank += drank;
      lastDrink = drank;
      drinksToday++;
      lastDrinkTime = millis();

      Serial.print("Drank: ");
      Serial.print(drank, 0);
      Serial.println(" mL");
    }

    referenceWater = water;
  }
}

void updateSensor() {
  unsigned long now = millis();

  // Start the first five-second averaging window.
  if (averageWindowStartedAt == 0) {
    averageWindowStartedAt = now;
  }

  // Collect one reading approximately every 100 ms.
  if (now - lastSampleAt >= SAMPLE_INTERVAL) {
    lastSampleAt = now;

    rawTotal += tx_rx();
    rawCount++;
  }

  // Do not update the displayed value until five seconds have passed.
  if (now - averageWindowStartedAt < AVERAGE_WINDOW) {
    return;
  }

  if (rawCount == 0) {
    averageWindowStartedAt = now;
    return;
  }

  uint32_t completedSampleCount = rawCount;

  // Average every reading collected during the complete five-second window.
  rawReading = static_cast<long>(rawTotal / rawCount);

  rawTotal = 0;
  rawCount = 0;
  averageWindowStartedAt = now;

  if (calibrated) {
    float newWater = rawToMl(rawReading);

    if (!filterStarted) {
      filteredWater = newWater;
      filterStarted = true;
    } else {
      // Additional smoothing between consecutive five-second averages.
      filteredWater =
        0.25f * newWater +
        0.75f * filteredWater;
    }

    water = filteredWater;

    processWaterTracking();
  }

  Serial.print("5-second average raw: ");
  Serial.print(rawReading);

  if (calibrated) {
    Serial.print("   Water: ");
    Serial.print(water, 0);
    Serial.print(" mL");
  }

  Serial.print("   Samples averaged: ");
  Serial.println(completedSampleCount);
}

// ============================================================
// WEB ROUTES
// ============================================================

void handleData() {
  float percent = totalDrank / DAILY_GOAL_ML * 100.0f;

  if (percent < 0.0f) percent = 0.0f;
  if (percent > 100.0f) percent = 100.0f;

  unsigned long secondsSinceDrink =
    trackingStarted
      ? (millis() - lastDrinkTime) / 1000UL
      : 0;

  long secondsLeft = 3600L - (long)secondsSinceDrink;
  if (secondsLeft < 0) secondsLeft = 0;

  unsigned long reminderMinutes =
    (secondsLeft + 59UL) / 60UL;

  char json[320];

  snprintf(
    json,
    sizeof(json),
    "{\"calibrated\":%s,"
    "\"raw\":%ld,"
    "\"water\":%.0f,"
    "\"total\":%.0f,"
    "\"last\":%.0f,"
    "\"count\":%d,"
    "\"percent\":%.1f,"
    "\"seconds\":%lu,"
    "\"reminder\":%lu,"
    "\"span\":%ld}",
    calibrated ? "true" : "false",
    rawReading,
    water,
    totalDrank,
    lastDrink,
    drinksToday,
    percent,
    secondsSinceDrink,
    reminderMinutes,
    fullRaw - emptyRaw
  );

  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "application/json", json);
}

void handleCalibrate() {
  if (!server.hasArg("empty") || !server.hasArg("full")) {
    server.send(
      400,
      "application/json",
      "{\"message\":\"Enter both values.\"}"
    );
    return;
  }

  long newEmpty = server.arg("empty").toInt();
  long newFull = server.arg("full").toInt();

  if (newEmpty <= 0 || newFull <= 0) {
    server.send(
      400,
      "application/json",
      "{\"message\":\"Values must be positive numbers.\"}"
    );
    return;
  }

  if (labs(newFull - newEmpty) < 20) {
    server.send(
      400,
      "application/json",
      "{\"message\":\"Empty and full values are too close.\"}"
    );
    return;
  }

  emptyRaw = newEmpty;
  fullRaw = newFull;
  calibrated = true;

  filterStarted = false;
  trackingStarted = false;

  Serial.print("Empty raw: ");
  Serial.println(emptyRaw);

  Serial.print("Full raw: ");
  Serial.println(fullRaw);

  server.send(
    200,
    "application/json",
    "{\"message\":\"Calibration accepted.\"}"
  );
}

void handleRecalibrate() {
  calibrated = false;
  filterStarted = false;
  trackingStarted = false;
  water = 0.0f;

  server.send(
    200,
    "application/json",
    "{\"message\":\"Ready to recalibrate.\"}"
  );
}

void handleReset() {
  totalDrank = 0.0f;
  lastDrink = 0.0f;
  drinksToday = 0;
  lastDrinkTime = millis();

  server.send(
    200,
    "application/json",
    "{\"message\":\"Total reset.\"}"
  );
}

// ============================================================
// SETUP / LOOP
// ============================================================

void setup() {
  pinMode(tx_pin, OUTPUT);
  pinMode(analog_pin, INPUT);
  digitalWrite(tx_pin, LOW);

  Serial.begin(115200);
  delay(1200);

  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.setAutoReconnect(true);

  WiFi.begin(WIFI_NAME, WIFI_PASSWORD);

  Serial.print("Connecting");

  unsigned long started = millis();

  while (
    WiFi.status() != WL_CONNECTED &&
    millis() - started < 30000UL
  ) {
    delay(400);
    Serial.print(".");
  }

  Serial.println();

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi failed. Check name/password.");
    return;
  }

  Serial.print("Open: http://");
  Serial.println(WiFi.localIP());

  server.on("/", HTTP_GET, []() {
    server.sendHeader("Cache-Control", "no-store");
    server.send_P(200, "text/html", PAGE);
  });

  server.on("/data", HTTP_GET, handleData);
  server.on("/calibrate", HTTP_POST, handleCalibrate);
  server.on("/recalibrate", HTTP_POST, handleRecalibrate);
  server.on("/reset", HTTP_POST, handleReset);

  server.on("/ping", HTTP_GET, []() {
    server.send(200, "text/plain", "OK");
  });

  server.begin();

  averageWindowStartedAt = millis();
  lastSampleAt = millis();

  Serial.println("Server ready");
  Serial.println("Sensor values update once every 5 seconds.");
}

void loop() {
  server.handleClient();
  updateSensor();
  delay(1);
}
