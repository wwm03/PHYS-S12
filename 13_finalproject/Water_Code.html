class CapacitiveSensor {
 private:
  int analogPin;
  int txPin;
  int numberOfSamples;

 public:
  CapacitiveSensor(int analogPin, int txPin, int samples = 100) {
    this->analogPin = analogPin;
    this->txPin = txPin;
    this->numberOfSamples = samples;
  }

  void begin() {
    pinMode(analogPin, INPUT);
    pinMode(txPin, OUTPUT);
    digitalWrite(txPin, LOW);
  }

  long read() {
    long sum = 0;

    for (int i = 0; i < numberOfSamples; i++) {
      digitalWrite(txPin, HIGH);
      delayMicroseconds(100);
      int readHigh = analogRead(analogPin);

      digitalWrite(txPin, LOW);
      delayMicroseconds(100);
      int readLow = analogRead(analogPin);

      sum += readHigh - readLow;
    }

    return sum;
  }
};

float loadRawToMl(long raw) {
  float ml = LOAD_EMPTY_ML +
    (float)(raw - LOAD_EMPTY_RAW) *
    (LOAD_FULL_ML - LOAD_EMPTY_ML) /
    (float)(LOAD_FULL_RAW - LOAD_EMPTY_RAW);

  if (ml < LOAD_EMPTY_ML) ml = LOAD_EMPTY_ML;
  if (ml > LOAD_FULL_ML) ml = LOAD_FULL_ML;
  return ml;
}

MoistureState classifyCapValue(int level, long value) {
  capAssistedByLoad[level] = false;
  if (!capDryCalibrated) return MOISTURE_UNKNOWN;

  MoistureState normalState = value < capWetThreshold[level]
    ? MOISTURE_DRY
    : MOISTURE_WET;
  if (
    loadBand >= 0 &&
    value >= capWetThreshold[level] - CAP_LOAD_ASSIST_TOLERANCE &&
    value <= capWetThreshold[level] + CAP_LOAD_ASSIST_TOLERANCE
  ) {
    MoistureState loadExpectedState = loadBand > level
      ? MOISTURE_WET
      : MOISTURE_DRY;

    capAssistedByLoad[level] = loadExpectedState != normalState;
    return loadExpectedState;
  }

  return normalState;
}

int determineCapLevel() {

  if (capState[0] == MOISTURE_UNKNOWN) return -1;
  if (capState[0] == MOISTURE_DRY) return 0;

  if (capState[1] == MOISTURE_UNKNOWN) return -1;
  if (capState[1] == MOISTURE_DRY) return 1;

  if (capState[2] == MOISTURE_UNKNOWN) return -1;
  if (capState[2] == MOISTURE_DRY) return 2;

  return 3;
}

int determineLoadBand(long raw) {
  if (
    raw >= LOAD_BAND_MIN[0] - LOAD_ENDPOINT_TOLERANCE &&
    raw <= LOAD_BAND_MAX[0]
  ) {
    return 0;
  }

  if (
    raw >= LOAD_BAND_MIN[3] &&
    raw <= LOAD_BAND_MAX[3] + LOAD_ENDPOINT_TOLERANCE
  ) {
    return 3;
  }

  for (int band = 1; band < 3; band++) {
    if (raw >= LOAD_BAND_MIN[band] && raw <= LOAD_BAND_MAX[band]) {
      return band;
    }
  }
  return -1;
}

const char* moistureName(MoistureState state) {
  if (state == MOISTURE_WET) return "Wet";
  if (state == MOISTURE_DRY) return "Dry";
  return "Transition";
}

    default: return "Transition";
  }

    case 1: return "Stage 1–2 transition";
    case 2: return "Stage 2 band";
    case 3: return "Stage 3 band";
    default: return "Outside ranges";
  }
}

String expectedRangeName(int level) {
  if (level < 0 || level > 3) return "Waiting";

  long minimum = LOAD_BAND_MIN[level];
  long maximum = LOAD_BAND_MAX[level];

  if (level == 0) minimum -= LOAD_ENDPOINT_TOLERANCE;
  if (level == 3) maximum += LOAD_ENDPOINT_TOLERANCE;

  return String(minimum) + "–" + String(maximum);
}

void saveFirmWater(float waterMl) {
  lastFirmWater = waterMl;
  referenceWater = waterMl;
  firmWaterReady = true;
  lastFirmRefreshAt = millis();
}

void refreshFirmWaterIfDue() {
  if (
    !trackingStarted ||
    !firmWaterReady ||
    drinkSettling ||
    millis() - lastFirmRefreshAt < FIRM_REFRESH_INTERVAL
  ) {
    return;
  }

  if (acceptedWater + FIRM_REFRESH_NOISE_ML >= lastFirmWater) {
    saveFirmWater(acceptedWater);
  }
}

void beginDrinkSettling(unsigned long now) {
  if (drinkSettling || !firmWaterReady) return;

  waterBeforePossibleDrink = lastFirmWater;
  drinkSettling = true;
  drinkSettleStartedAt = now;
  fastDropSampleCount = 0;
  lastEvent = "Fast water drop detected; saved firm level " +
    String(waterBeforePossibleDrink, 0) + " mL and started a fresh 5-second average";
}

void startOrResetBaseline(const String& message) {
  if (!sensorsAgree) {
    lastEvent = "Cannot set baseline until the sensors agree";
    return;
  }

  saveFirmWater(acceptedWater);
  drinkSettling = false;
  fastDropSampleCount = 0;
  lastDrinkTime = millis();
  trackingStarted = true;
  lastEvent = message;
}

    return;
  }

  float change = acceptedWater - referenceWater;

  if (change >= REFILL_THRESHOLD_ML) {
    saveFirmWater(acceptedWater);
    lastEvent = "Refill detected: +" + String(change, 0) + " mL";
    return;
  }
}

void evaluateCompletedAverages() {
  bool sensorsPreviouslyAgreed = sensorsAgree;

  loadBand = loadCellReady ? determineLoadBand(loadRaw) : -1;
  loadMl = loadCellReady ? loadRawToMl(loadRaw) : 0.0f;

  for (int i = 0; i < 3; i++) {
    capState[i] = classifyCapValue(i, capRaw[i]);
  }

  if (!capDryCalibrated) {
    capLevel = -1;
    sensorsAgree = false;
    lastEvent = "Press Calibrate dry sensors before tracking";
    return;
  }

  capLevel = determineCapLevel();

  sensorsAgree =
    loadCellReady &&
    capLevel >= 0 &&
    loadBand >= 0 &&
    capLevel == loadBand;

  if (
    trackingStarted &&
    firmWaterReady &&
    loadCellReady &&
    !drinkSettling &&
    lastFirmWater - loadMl >= DRINK_THRESHOLD_ML
  ) {
    beginDrinkSettling(millis());
    return;
  }

  if (drinkSettling) {
    unsigned long settleElapsed = millis() - drinkSettleStartedAt;

    if (settleElapsed < DRINK_SETTLE_TIME) {
      unsigned long secondsLeft =
        (DRINK_SETTLE_TIME - settleElapsed + 999UL) / 1000UL;
      lastEvent = "Possible drink; settling for " + String(secondsLeft) + " more seconds";
      return;
    }

    if (!sensorsAgree) {
      lastEvent = "Five-second wait complete; waiting for capacitance and load cell to agree";
      return;
    }

    float settledWater = loadMl;
    float drank = waterBeforePossibleDrink - settledWater;

    acceptedWater = settledWater;
    filteredWater = settledWater;
    filterStarted = true;
    drinkSettling = false;
    fastDropSampleCount = 0;

    if (drank >= DRINK_THRESHOLD_ML && drank <= LOAD_FULL_ML) {
      saveFirmWater(settledWater);
      totalDrank += drank;
      lastDrink = drank;
      drinksToday++;
      lastDrinkTime = millis();
      lastEvent = "Drink detected from saved firm level: " + String(drank, 0) + " mL";
    } else if (-drank >= REFILL_THRESHOLD_ML) {
      float refilled = -drank;
      saveFirmWater(settledWater);
      lastEvent = "Refill detected during settling: +" + String(refilled, 0) + " mL";
    } else if (drank < DRINK_THRESHOLD_ML) {
      lastEvent = "Change was below 50 mL; previous firm level was kept";
    } else {
      saveFirmWater(settledWater);
      lastEvent = "Settled drop exceeded bottle capacity; no drink recorded";
    }

    return;
  }

  if (!sensorsAgree) {
    if (!loadCellReady) {
      lastEvent = "Waiting for HX711 load-cell data";
    } else if (capLevel < 0) {
      lastEvent = "A capacitance value is in its transition zone";
    } else if (loadBand < 0) {
      lastEvent = "Load-cell raw value is outside the calibrated ranges";
    } else {
      lastEvent = "Cross-check mismatch; water tracking is paused";
    }
    return;
  }

  if (!sensorsPreviouslyAgreed && trackingStarted) {
    lastEvent = "Sensor agreement restored; tracking resumed";
  }

  if (!filterStarted) {
    filteredWater = loadMl;
    filterStarted = true;
  } else {
    filteredWater = 0.35f * loadMl + 0.65f * filteredWater;
  }

  acceptedWater = filteredWater;
  processWaterTracking();
  refreshFirmWaterIfDue();
}

void updateSensors() {
  unsigned long now = millis();

  if (now - lastCapSampleAt >= CAP_SAMPLE_INTERVAL) {
    lastCapSampleAt = now;

    for (int i = 0; i < 3; i++) {
      capTotals[i] += levelSensors[i]->read();
    }
    capSampleCount++;
  }

  if (loadCell.is_ready()) {
    long liveLoadRaw = loadCell.read();
    loadTotal += liveLoadRaw;
    loadSampleCount++;

    if (trackingStarted && firmWaterReady && !drinkSettling) {
      float liveLoadMl = loadRawToMl(liveLoadRaw);

      if (lastFirmWater - liveLoadMl >= DRINK_THRESHOLD_ML) {
        if (fastDropSampleCount < FAST_DROP_CONFIRM_SAMPLES) {
          fastDropSampleCount++;
        }
      } else {
        fastDropSampleCount = 0;
      }

      if (fastDropSampleCount >= FAST_DROP_CONFIRM_SAMPLES) {
        beginDrinkSettling(now);

        for (int i = 0; i < 3; i++) {
          capTotals[i] = 0;
        }
        capSampleCount = 0;
        loadTotal = 0;
        loadSampleCount = 0;
        averageWindowStartedAt = now;
        lastCapSampleAt = now;
        return;
      }
    }
  }

  if (now - averageWindowStartedAt < AVERAGE_WINDOW) return;

  if (capSampleCount == 0) {
    averageWindowStartedAt = now;
    return;
  }

  for (int i = 0; i < 3; i++) {
    capRaw[i] = static_cast(capTotals[i] / capSampleCount);
    capTotals[i] = 0;
  }
  sensorAverageReady = true;

  if (loadSampleCount > 0) {
    loadRaw = static_cast(loadTotal / loadSampleCount);
    loadCellReady = true;
  } else {
    loadCellReady = false;
  }

  uint32_t completedCapSamples = capSampleCount;
  uint32_t completedLoadSamples = loadSampleCount;

  capSampleCount = 0;
  loadTotal = 0;
  loadSampleCount = 0;
  averageWindowStartedAt = now;

  evaluateCompletedAverages();

  Serial.println("\n--- Five-second sensor average ---");
  for (int i = 0; i < 3; i++) {
    Serial.print("Level ");
    Serial.print(i + 1);
    Serial.print(": ");
    Serial.print(capRaw[i]);
    Serial.print(" (");
    Serial.print(moistureName(capState[i]));
    if (capAssistedByLoad[i]) {
      Serial.print(", load assisted");
    }
    Serial.println(")");
  }

  Serial.print("Load cell raw: ");
  if (loadCellReady) {
    Serial.print(loadRaw);
    Serial.print(" = ");
    Serial.print(loadMl, 1);
    Serial.println(" mL");
  } else {
    Serial.println("not ready");
  }

  Serial.print("Cap level: ");
  Serial.println(capLevelName(capLevel));
  Serial.print("Load band: ");
  Serial.println(loadBandName(loadBand));
  Serial.print("Cross-check: ");
  Serial.println(sensorsAgree ? "VERIFIED" : "MISMATCH / WAITING");
  Serial.print("Samples: capacitance=");
  Serial.print(completedCapSamples);
  Serial.print(", load cell=");
  Serial.println(completedLoadSamples);
}

void handleData() {
  float percent = totalDrank / DAILY_GOAL_ML * 100.0f;
  if (percent < 0.0f) percent = 0.0f;
  if (percent > 100.0f) percent = 100.0f;

  unsigned long secondsSinceDrink = trackingStarted
    ? (millis() - lastDrinkTime) / 1000UL
    : 0;

  unsigned long secondsLeft = secondsSinceDrink >= REMINDER_TIME / 1000UL
    ? 0
    : REMINDER_TIME / 1000UL - secondsSinceDrink;

  unsigned long reminderMinutes = (secondsLeft + 59UL) / 60UL;

  bool useLevel2 = capState[0] == MOISTURE_WET;
  bool useLevel3 = useLevel2 && capState[1] == MOISTURE_WET;

  String json;
  json.reserve(1150);
  json += "{";
  json += "\"averageReady\":" + String(sensorAverageReady ? "true" : "false");
  json += ",\"capCalibrated\":" + String(capDryCalibrated ? "true" : "false");
  json += ",\"loadReady\":" + String(loadCellReady ? "true" : "false");
  json += ",\"loadRaw\":" + String(loadRaw);
  json += ",\"loadMl\":" + String(loadMl, 1);
  json += ",\"water\":" + String(acceptedWater, 1);
  json += ",\"firmWater\":" + String(lastFirmWater, 1);
  json += ",\"firmReady\":" + String(firmWaterReady ? "true" : "false");
  json += ",\"firmAge\":" + String(
    firmWaterReady ? (millis() - lastFirmRefreshAt) / 1000UL : 0
  );
  json += ",\"total\":" + String(totalDrank, 1);
  json += ",\"last\":" + String(lastDrink, 1);
  json += ",\"count\":" + String(drinksToday);
  json += ",\"percent\":" + String(percent, 1);
  json += ",\"seconds\":" + String(secondsSinceDrink);
  json += ",\"reminder\":" + String(reminderMinutes);
  json += ",\"settling\":" + String(drinkSettling ? "true" : "false");
  json += ",\"verified\":" + String(sensorsAgree ? "true" : "false");
  json += ",\"capLevel\":" + String(capLevel);
  json += ",\"capLevelName\":\"" + String(capLevelName(capLevel)) + "\"";
  json += ",\"loadBand\":" + String(loadBand);
  json += ",\"loadBandName\":\"" + String(loadBandName(loadBand)) + "\"";
  json += ",\"expectedRange\":\"" + expectedRangeName(capLevel) + "\"";
  json += ",\"s1raw\":" + String(capRaw[0]);
  json += ",\"s2raw\":" + String(capRaw[1]);
  json += ",\"s3raw\":" + String(capRaw[2]);
  json += ",\"s1base\":" + String(capDryBaseline[0]);
  json += ",\"s2base\":" + String(capDryBaseline[1]);
  json += ",\"s3base\":" + String(capDryBaseline[2]);
  json += ",\"s1threshold\":" + String(capWetThreshold[0]);
  json += ",\"s2threshold\":" + String(capWetThreshold[1]);
  json += ",\"s3threshold\":" + String(capWetThreshold[2]);
  json += ",\"s1state\":" + String((int)capState[0]);
  json += ",\"s2state\":" + String((int)capState[1]);
  json += ",\"s3state\":" + String((int)capState[2]);
  json += ",\"s1assisted\":" + String(capAssistedByLoad[0] ? "true" : "false");
  json += ",\"s2assisted\":" + String(capAssistedByLoad[1] ? "true" : "false");
  json += ",\"s3assisted\":" + String(capAssistedByLoad[2] ? "true" : "false");
  json += ",\"s2used\":" + String(useLevel2 ? "true" : "false");
  json += ",\"s3used\":" + String(useLevel3 ? "true" : "false");
  json += ",\"event\":\"" + lastEvent + "\"";
  json += "}";

  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "application/json", json);
}

void handleCalibrateDry() {
  if (!sensorAverageReady) {
    server.send(
      409,
      "application/json",
      "{\"message\":\"Wait for the first five-second sensor average.\"}"
    );
    return;
  }

  for (int i = 0; i < 3; i++) {
    capDryBaseline[i] = capRaw[i];
    capWetThreshold[i] = capRaw[i] + CAP_WET_OFFSET[i];
    capState[i] = MOISTURE_DRY;
    capAssistedByLoad[i] = false;
  }

  capDryCalibrated = true;
  capLevel = 0;
  sensorsAgree = false;
  filterStarted = false;
  trackingStarted = false;
  drinkSettling = false;
  firmWaterReady = false;
  fastDropSampleCount = 0;
  acceptedWater = 0.0f;
  filteredWater = 0.0f;
  referenceWater = 0.0f;
  lastFirmWater = 0.0f;
  lastDrinkTime = millis();
  lastEvent = "Dry calibration saved; waiting for the next sensor average";

  Serial.println("\nDry capacitance calibration saved:");
  for (int i = 0; i < 3; i++) {
    Serial.print("Level ");
    Serial.print(i + 1);
    Serial.print(" baseline=");
    Serial.print(capDryBaseline[i]);
    Serial.print(", wet at/above ");
    Serial.println(capWetThreshold[i]);
  }

  server.send(
    200,
    "application/json",
    "{\"message\":\"Dry readings saved with Level 1 +3,000, Level 2 +10,000, and Level 3 +10,000.\"}"
  );
}

void handleReset() {
  totalDrank = 0.0f;
  lastDrink = 0.0f;
  drinksToday = 0;
  drinkSettling = false;
  fastDropSampleCount = 0;

  if (sensorsAgree) {
    saveFirmWater(acceptedWater);
    trackingStarted = true;
  } else {
    firmWaterReady = false;
    trackingStarted = false;
  }

  lastDrinkTime = millis();
  lastEvent = "Today's total was reset";
  server.send(200, "application/json", "{\"message\":\"Total reset.\"}");
}

void handleBaseline() {
  startOrResetBaseline("Baseline reset to the current verified water level");
  server.send(200, "application/json", "{\"message\":\"Baseline request processed.\"}");
}

void startWebServer() {
  server.on("/", HTTP_GET, []() {
    server.sendHeader("Cache-Control", "no-store");
    server.send_P(200, "text/html", PAGE);
  });

  server.on("/data", HTTP_GET, handleData);
  server.on("/calibrate-dry", HTTP_POST, handleCalibrateDry);
  server.on("/reset", HTTP_POST, handleReset);
  server.on("/baseline", HTTP_POST, handleBaseline);
  server.on("/ping", HTTP_GET, []() {
    server.send(200, "text/plain", "OK");
  });

  server.begin();
}

void printWebsiteAddress() {
  Serial.println();
  Serial.println("==================================================");
  Serial.println("        SMART WATER BOTTLE WEBSITE");
  Serial.print("COPY THIS LINK INTO YOUR BROWSER: http://");

  if (usingBackupHotspot) {
    Serial.print(WiFi.softAPIP());
  } else {
    Serial.print(WiFi.localIP());
  }

  Serial.println("/");
  Serial.println("==================================================");
  Serial.println();
}

void setup() {
  Serial.begin(115200);
  delay(1200);

  sensor1.begin();
  sensor2.begin();
  sensor3.begin();
  loadCell.begin(HX711_DT, HX711_SCK);

  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_NAME, WIFI_PASSWORD);

  Serial.print("Connecting to Wi-Fi");
  unsigned long connectionStarted = millis();

  while (
    WiFi.status() != WL_CONNECTED &&
    millis() - connectionStarted < 30000UL
  ) {
    delay(400);
    Serial.print(".");
  }

  Serial.println();

  if (WiFi.status() == WL_CONNECTED) {
    usingBackupHotspot = false;
  } else {
    WiFi.disconnect(true);
    WiFi.mode(WIFI_AP);
    WiFi.softAP(AP_NAME, AP_PASSWORD);
    usingBackupHotspot = true;

    Serial.println("Normal Wi-Fi failed; fallback hotspot started.");
    Serial.print("Join network: ");
    Serial.println(AP_NAME);
  }

  startWebServer();
  printWebsiteAddress();
  lastLinkPrintAt = millis();

  averageWindowStartedAt = millis();
  lastCapSampleAt = millis();

  Serial.println("Server ready. Sensor averages update every five seconds.");
}

void loop() {
  server.handleClient();
  updateSensors();

  if (millis() - lastLinkPrintAt >= LINK_PRINT_INTERVAL) {
    lastLinkPrintAt = millis();
    printWebsiteAddress();
  }

  delay(1);
}
