String Inputword = "";
typedef struct {
  char message[80];
} datas;

datas data;

uint8_t morseReceiverMAC[] = {0x94, 0xA9, 0x90, 0x6B, 0x5A, 0x9C};
void onDataRecv(const esp_now_recv_info *info, const uint8_t *incomingData, int len){
  memcpy(
    &data,
    incomingData,
    sizeof(data)
  );
  Inputword = "";
   Serial.println("Enter A Word To Decode: ");
}
void setup() {
  Serial.begin(9600);
  Serial.println("Enter A Word To Decode: ");
  WiFi.mode(WIFI_STA);
  if (esp_now_init() != ESP_OK) {
    return;
  }

  esp_now_peer_info_t peerInfo = {};
  memcpy(peerInfo.peer_addr, morseReceiverMAC, 6);
  peerInfo.channel = 0;
  peerInfo.encrypt = false;
  esp_now_add_peer(&peerInfo);
  // put your setup code here, to run once:
  esp_now_register_recv_cb(onDataRecv);

}

void loop() {
  // put your main code here, to run repeatedly:
  if (Serial.available() > 0 && Inputword == "") {
    Inputword = Serial.readStringUntil('\n');
    Inputword.trim();
    Serial.println("Sending " + String(Inputword));
    strcpy(data.message, Inputword.c_str());
    esp_now_send(morseReceiverMAC,(uint8_t*)&data, sizeof(data));
  }
}
